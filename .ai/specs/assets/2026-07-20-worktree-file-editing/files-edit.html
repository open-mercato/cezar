export function resolveCapabilities(env: NodeJS.ProcessEnv, bindHost?: string): Capabilities {
  return {
    localHandoff: env.CEZ_REMOTE !== '1' && isLoopbackHost(bindHost),
    fileEdit: env.CEZ_REMOTE !== '1' && isLoopbackHost(bindHost) && env.CEZ_NO_FILE_EDIT !== '1',
    // Deliberately not re-derived here: RunManager enforces the same predicate,
    // and two spellings of "is the inbox on" would eventually disagree.
    followups: followupsEnabled(env),
  };
}

/** Loopback only — a bind on 0.0.0.0 is reachable from the network. */
function isLoopbackHost(host?: string): boolean {
  if (!host) return true;
  return host === '127.0.0.1' || host === '::1' || host === 'localhost';
}
